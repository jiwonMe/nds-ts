START WITH A FEW LINES
var x = 120;
var game: NDSGame = {
  update: function () {
    x += nds.input.dx() * 2;
  },
  draw: function () {
    nds.top.clear(0xFFFFFF);
    nds.top.sprite('orb', x, 138);
  }
};
